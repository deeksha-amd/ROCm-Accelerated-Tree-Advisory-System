# Dependencies for the tree advisory pipeline.
#
# torch is NOT listed: the rocm/pytorch image already ships it built against
# ROCm, and installing from PyPI would replace it with a CUDA build that
# cannot see the MI210.
#
#   pip install -r requirements.txt

# ── Raster and geospatial ─────────────────────────────────────────────
rasterio>=1.3          # reads every climate / soil / terrain GeoTIFF
affine>=2.4            # grid transforms; pulled in by rasterio but used directly
shapely>=2.0           # country polygons for the coverage test

# ── Numerics ──────────────────────────────────────────────────────────
numpy>=1.24            # AUC, TSS and Boyce are implemented directly in numpy

# ── Data and network ──────────────────────────────────────────────────
pandas>=2.0            # only xgboost_training.py; skip if not running that
requests>=2.28         # GBIF REST client (pygbif is deliberately not used)
tqdm>=4.65             # download and aggregation progress bars

# ── Web advisory ──────────────────────────────────────────────────────
flask>=3.0             # advisory_app.py only; not needed for training

# ── Optional: the XGBoost baseline only ───────────────────────────────
# The deep SDM needs neither of these. Uncomment only to run
# xgboost_training.py, which is the legacy comparison baseline.
# scikit-learn>=1.3
# xgboost>=2.0
